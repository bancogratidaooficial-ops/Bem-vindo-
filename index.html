// BANCO GRATIDÃO v2.0 - RECEPTOR DO BOTÃO FÍSICO DO BARCO function doPost(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Página1"); var lr = sheet.getLastRow() + 1; var dados = JSON.parse(e.postData.contents); // IA 2.0 classifica sozinha var desc = dados.descricao || "Combustível - D"; var cat = "Outros"; if (desc.toLowerCase().includes("comb") || desc.toLowerCase().includes("diesel")) cat = "Combustível"; if (desc.toLowerCase().includes("manut")) cat = "Manutenção"; sheet.getRange(lr, 1).setValue(new Date()); sheet.getRange(lr, 2).setValue(dados.tipo || "SAÍDA"); sheet.getRange(lr, 3).setValue(desc); sheet.getRange(lr, 4).setValue(cat); sheet.getRange(lr, 5).setValue(dados.valor || 180); sheet.getRange(lr, 6).setValue("Botão Barco Flic"); sheet.getRange(lr, 7).setFormula("=G"+(lr-1)+"+IF(B"+lr+"=\"ENTRADA\",E"+lr+",-E"+lr+")"); return ContentService.createTextOutput("OK - Gasto lançado: R$ " + dados.valor); } // GERA O LINK DO SEU BANCO function getLinkBanco() { var url = ScriptApp.getService().getUrl(); Logger.log(url); Browser.msgBox("Seu link é: " + url + " - Copia esse link pro Flic!"); }
